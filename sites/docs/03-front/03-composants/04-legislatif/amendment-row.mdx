import { AmendmentRow } from "@democracy-watch/ui";
import { amendments, articles, texts } from "@democracy-watch/data";

export const toResult = (number) => {
  const amendment = amendments.find((item) => item.number === number);
  return {
    ...amendment,
    text: texts.find((text) => text.id === amendment.textId),
    article: articles.find((article) => article.id === amendment.articleId) ?? null,
  };
};

export const ce31 = toResult("CE 31");
