import { commissions, texts } from "@democracy-watch/data";
import { TextsTable } from "@democracy-watch/ui";

export const commissionName = (id) => commissions.find((c) => c.id === id)?.name ?? id;

## Import

```tsx
import { TextsTable } from "@democracy-watch/ui";
```

## Utilisation de base

<Preview>
  <TextsTable texts={texts} commissionName={commissionName} selected={[]} onToggle={() => {}} />
</Preview>

```tsx
const commissionName = (id) => commissions.find((c) => c.id === id)?.name ?? id;

<TextsTable texts={texts} commissionName={commissionName} selected={[]} onToggle={() => {}} />
```
